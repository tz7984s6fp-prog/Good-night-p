💗 Good Night Princess 👑

import turtle

screen = turtle.Screen()
screen.bgcolor("black")
screen.title("Good Night Princess")
screen.setup(width=800, height=800)

t = turtle.Turtle()
t.speed(0)
t.hideturtle()

# Pink heart
t.penup()
t.goto(0, -260)
t.setheading(140)
t.pendown()
t.color("#ff69b4")
t.fillcolor("#ff69b4")

t.begin_fill()
t.forward(240)
t.circle(-120, 200)
t.left(120)
t.circle(-120, 200)
t.forward(240)
t.end_fill()

# Text
t.penup()
t.goto(0, -15)
t.color("white")
t.write("good night", align="center",
        font=("Arial", 32, "bold"))

t.goto(0, -70)
t.write("princess", align="center",
        font=("Arial", 34, "bold"))

# Hearts
t.goto(-170, 55)
t.color("#ff1493")
t.write("♥", align="center", font=("Arial", 28, "bold"))

t.goto(170, 55)
t.write("♥", align="center", font=("Arial", 28, "bold"))

# Crown
t.goto(0, 180)
t.color("#ffd700")
t.write("♕", align="center", font=("Arial", 45, "bold"))

turtle.done()
